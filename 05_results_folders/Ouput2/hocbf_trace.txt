# Diagnostic wrapper: keeps the original filter policy and QP unchanged.
# Records branch, fresh solve status, fallback and slack for EACH call.
class InstrumentedHOCBF:
    def __init__(self, original):
        self.base = original
        self.last = {}

    def __getattr__(self, name):
        return getattr(self.base, name)

    def reset_stats(self):
        self.base.reset_stats()
        self.last = {}

    def filter(self, x, tau_rl):
        # Re-evaluate only the documented fast-path condition; never alter the action.
        lf, lg, psi1, _ = self.base._compute_lie(x)
        pre = float(lf + lg @ tau_rl + self.base.alpha2 * psi1)
        bypass = bool(pre >= 0 and psi1 >= 0)
        self.last = dict(qp_attempted=not bypass, qp_solved=False,
                         qp_failed=False, fallback=False, qp_status='not_attempted',
                         slack_value=np.nan, slack_used=False, bypass=bypass)
        # Prevent stale OSQP values/status being mistaken for this call.
        # Existing filter catches exceptions and falls back; no solver monkeypatching.
        # Replace only solve on this instance during one synchronous filter call.
        # This records whether OSQP was called and whether it returned or raised.
        qp = self.base._qp
        orig_solve = qp.solve
        outcome = {'called': False, 'raised': False, 'status': None, 'has_tau': False}
        def observed_solve(*args, **kwargs):
            outcome['called'] = True
            try:
                result = orig_solve(*args, **kwargs)
                outcome['status'] = qp.status
                outcome['has_tau'] = self.base._tau.value is not None
                return result
            except Exception:
                outcome['raised'] = True
                outcome['status'] = qp.status
                raise
        qp.solve = observed_solve
        try:
            tau = self.base.filter(x, tau_rl)
        finally:
            qp.solve = orig_solve
        if not bypass:
            valid = outcome['status'] in (self.base.cp.OPTIMAL, self.base.cp.OPTIMAL_INACCURATE)
            solved = bool(outcome['called'] and not outcome['raised'] and valid and outcome['has_tau'])
            self.last.update(qp_attempted=outcome['called'], qp_status=str(outcome['status']),
                             qp_solved=solved, qp_failed=not solved, fallback=not solved)
            if solved and self.base._slack.value is not None:
                sv = float(self.base._slack.value[0])
                self.last.update(slack_value=sv, slack_used=bool(sv > 1e-3))
        return tau

    def mpc_control(self):
        t0 = time.perf_counter()

        c = abs(self.state[0]) > np.pi / 4
        return c, c or self._steps >= self.MAX_STEPS

    # â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
    def _compute_lie(self, x: np.ndarray):
        """
        ظ…ط­ط§ط³ط¨ظ‡ Lf_دˆâ‚پطŒ Lg_دˆâ‚پ ظˆ دˆâ‚پ.

        x = [د†, د†ج‡, خ´_f, خ´_r, خ´ج‡_f, خ´ج‡_r]
        """
        phi, phi_dot = x[0], x[1]
        ddot_f, ddot_r = x[4], x[5]

        # دˆâ‚€ ظˆ دˆâ‚پ
        psi0 = CRASH_ANGLE ** 2 - phi ** 2
        psi1 = -2.0 * phi * phi_dot + self.alpha1 * psi0

        # f_phi(x): drift term roll acceleration (ط¨ط¯ظˆظ† ع©ظ†طھط±ظ„)
        cos_phi = np.cos(phi)
        sin_phi = np.sin(phi)
        f_phi = (
            self.mgh * sin_phi
            - self.IwO * (ddot_f - ddot_r) * cos_phi
        ) / self.Iphi

        # g_phi: âˆ‚د†جˆ/âˆ‚د„
        # د„ â†’ خ´جˆ â†’ خ´ج‡_next (غŒظ‡ step طھط£ط®غŒط±)
        # âˆ‚خ´ج‡_f_next/âˆ‚د„_f â‰ˆ Ts / I_d   (ط§ط² ط¯غŒظ†ط§ظ…غŒع© ع¯غŒظ…ط¨ط§ظ„)
        # د†جˆ = (... âˆ’ Iwخ©آ·(خ´ج‡_fâˆ’خ´ج‡_r)آ·cos(د†)) / I_phi
        # âˆ‚د†جˆ/âˆ‚د„_f â‰ˆ âˆ’Iwخ©آ·cos(د†)آ·(Ts/I_d) / I_phi
        # âˆ‚د†جˆ/âˆ‚د„_r â‰ˆ +Iwخ©آ·cos(د†)آ·(Ts/I_d) / I_phi
        k = self.IwO * cos_phi * Ts / (self.Iphi * self.Id)
        g_phi = np.array([-k, +k])   # [âˆ‚د†جˆ/âˆ‚د„_f, âˆ‚د†جˆ/âˆ‚د„_r]

        # دˆج‡â‚پ = Lf_دˆâ‚پ + Lg_دˆâ‚پ آ· د„
        # Lf_دˆâ‚پ = âˆ’2د†ج‡آ² âˆ’ 2د†آ·f_phi âˆ’ 2خ±â‚پآ·د†آ·د†ج‡
        Lf_psi1 = (
            -2.0 * phi_dot ** 2
            - 2.0 * phi * f_phi
            - 2.0 * self.alpha1 * phi * phi_dot
        )
        # Lg_دˆâ‚پ = âˆ’2د† آ· g_phi
        Lg_psi1 = -2.0 * phi * g_phi

        return Lf_psi1, Lg_psi1, psi1, psi0

    # â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
    def filter(self, x: np.ndarray, tau_rl: np.ndarray) -> np.ndarray:
        """
        ط§ع¯ط± د„_rl ط´ط±ط· HOCBF ط±ط§ ط¨ط±ط¢ظˆط±ط¯ظ‡ ع©ظ†ط¯ â†’ ط¨ط¯ظˆظ† طھط؛غŒغŒط±.
        ظˆع¯ط±ظ†ظ‡ â†’ QP ط¨ط§ slack variable (ظ‡ظ…غŒط´ظ‡ feasible).

        Slack variable طھط¶ظ…غŒظ† ظ…غŒâ€Œع©ظ†ط¯ ع©ظ‡ QP ظ‡ط±ع¯ط² infeasible ظ†ظ…غŒâ€Œط´ظˆط¯.
        ط§ع¯ط± s > 0 ط¯ط± ط¬ظˆط§ط¨ â†’ ظ‚غŒط¯ CBF ط¨ط§ ط¬ط±غŒظ…ظ‡ خ»آ·sآ² relaxed ط´ط¯ظ‡.
        """
        self.n_total += 1

        Lf_psi1, Lg_psi1, psi1, psi0 = self._compute_lie(x)

        # ط¨ط±ط±ط³غŒ ط§غŒظ…ظ†غŒ
        cbf_val = Lf_psi1 + Lg_psi1 @ tau_rl + self.alpha2 * psi1
        if cbf_val >= 0 and psi1 >= 0:
            return np.clip(tau_rl, -TAU_MAX, TAU_MAX)

        # QP ط¨ط§ slack
        self._tau_rl.value  = tau_rl
        self._Lf_p.value    = np.array([Lf_psi1])
        self._Lg_p.value    = Lg_psi1
        self._psi1_p.value  = np.array([psi1])
        self._lam_p.value   = np.array([self.lambda_slack])

        try:
            self._qp.solve(
                solver=self.cp.OSQP,
                warm_start=True,
                eps_abs=1e-5, eps_rel=1e-5,
                verbose=False,
            )
            if (self._qp.status in (self.cp.OPTIMAL, self.cp.OPTIMAL_INACCURATE)
                    and self._tau.value is not None):
                self.n_filtered += 1
                # ط±ط¯غŒط§ط¨غŒ: ط¢غŒط§ slack ظپط¹ط§ظ„ ط´ط¯طں
                s_val = float(self._slack.value[0]) if self._slack.value is not None else 0.0
                if s_val > 1e-3:
                    self.n_slack_used += 1
                return np.clip(self._tau.value, -TAU_MAX, TAU_MAX)
        except Exception:
            pass

        # fallback ط§ط¶ط·ط±ط§ط±غŒ: ط¨ط§ slack QP ظ†ط¨ط§غŒط¯ ط¨ظ‡ ط§غŒظ†ط¬ط§ ط¨ط±ط³ط¯طŒ
        # ظˆظ„غŒ ط¨ط±ط§غŒ ظ‡ط± ط§ط­طھظ…ط§ظ„غŒ tau_rl clipâ€Œط´ط¯ظ‡ ط¨ط±ظ…غŒâ€Œع¯ط±ط¯ط§ظ†غŒظ….
        self.n_filtered += 1
        return np.clip(tau_rl, -TAU_MAX, TAU_MAX)

    def reset_stats(self):
        self.n_filtered   = 0
        self.n_total      = 0
        self.n_slack_used = 0

    @property
    def filter_rate(self) -> float:
        return self.n_filtered / self.n_total if self.n_total > 0 else 0.0

    @property
    def slack_rate(self) -> float:
        """ظ†ط³ط¨طھ ط¯ظپط¹ط§طھغŒ ع©ظ‡ slack ظپط¹ط§ظ„ ط´ط¯ (ظ‚غŒط¯ CBF relaxed ط´ط¯)."""
        return self.n_slack_used / self.n_total if self.n_total > 0 else 0.0
